// ── Main App ──────────────────────────────────────────────────────────────────
export default function App() {
  const [dark, setDark] = useState(false);
  const [page, setPage] = useState("home");
  const [mobileOpen, setMobileOpen] = useState(false);
  const [authTab, setAuthTab] = useState("login");
  const [sermonFilter, setSermonFilter] = useState("All");
  const [prayerText, setPrayerText] = useState("");
  const [prayerSent, setPrayerSent] = useState(false);
  const styleRef = useRef(null);

  useEffect(() => {
    if (!styleRef.current) {
      styleRef.current = document.createElement("style");
      document.head.appendChild(styleRef.current);
    }
    styleRef.current.textContent = getStyles(dark);
  }, [dark]);

  const navigate = (p) => { setPage(p); setMobileOpen(false); window.scrollTo(0, 0); };
  const navItems = ["Home", "Sermons", "Departments", "Events", "Devotionals", "Live", "Give"];

  const sendPrayer = () => { if (prayerText.trim()) { setPrayerSent(true); setPrayerText(""); setTimeout(() => setPrayerSent(false), 4000); } };

  return (
    <div className="app">
      {/* NAVBAR */}
      <nav className="nav">
        <button className="nav-brand" onClick={() => navigate("home")}>
          <div className="nav-logo">CS</div>
          <div className="nav-name">
            RCCG Covenant Sanctuary
            <span>Ibadan Province</span>
          </div>
        </button>

        <div className="nav-links">
          {navItems.map(n => (
            <button key={n} className={`nav-link ${page === n.toLowerCase() ? "active" : ""}`} onClick={() => navigate(n.toLowerCase())}>
              {n}
            </button>
          ))}
        </div>

        <div className="nav-actions">
          <button className="nav-live" onClick={() => navigate("live")}>
            <span style={{ width: 8, height: 8, borderRadius: "50%", background: "#fff", display: "inline-block", animation: "livepulse 1.5s infinite" }} />
            Join Live
          </button>
          <button className="icon-btn" onClick={() => setDark(!dark)}>
            <Icon name={dark ? "sun" : "moon"} size={16} />
          </button>
          <button className="icon-btn" onClick={() => navigate("account")}>
            <Icon name="user" size={16} />
          </button>
          <button className="icon-btn hamburger" onClick={() => setMobileOpen(!mobileOpen)}>
            <Icon name={mobileOpen ? "x" : "menu"} size={18} />
          </button>
        </div>
      </nav>

      {/* MOBILE MENU */}
      <div className={`mobile-menu ${mobileOpen ? "open" : ""}`}>
        {navItems.map(n => (
          <button key={n} className={`mobile-link ${page === n.toLowerCase() ? "active" : ""}`} onClick={() => navigate(n.toLowerCase())}>
            {n}
          </button>
        ))}
        <button className="mobile-live" onClick={() => navigate("live")}>🔴 Join Live Service</button>
      </div>

      {/* PAGES */}
      {page === "home" && <HomePage navigate={navigate} />}
      {page === "sermons" && <SermonsPage filter={sermonFilter} setFilter={setSermonFilter} />}
      {page === "departments" && <DepartmentsPage />}
      {page === "events" && <EventsPage />}
      {page === "devotionals" && <DevotionalsPage />}
      {page === "live" && <LivePage prayerText={prayerText} setPrayerText={setPrayerText} prayerSent={prayerSent} sendPrayer={sendPrayer} />}
      {page === "give" && <GivePage />}
      {page === "account" && <AccountPage authTab={authTab} setAuthTab={setAuthTab} />}

      {/* FOOTER (all pages) */}
      <Footer navigate={navigate} />
    </div>
  );
}
